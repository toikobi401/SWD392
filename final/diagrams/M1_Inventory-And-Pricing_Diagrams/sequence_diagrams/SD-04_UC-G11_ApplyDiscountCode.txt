@startuml
title SD-04 UC-G11 Apply Discount Code

actor "Guest" as G
boundary "BookingWizardPage" as UI
boundary "PromotionController" as C
entity "Promotion" as P
database "MongoDB" as DB

G -> UI : enter discount code
activate G
activate UI
UI -> C : check discount code
activate C
C -> C : validate request
C -> P : get promotion by code
activate P
P -> DB : find promotion
activate DB
DB --> P : promotion data
deactivate DB
P --> C : promotion
deactivate P
alt code is valid and applicable
C --> UI : discount amount and new total
UI --> G : show discounted total
else code invalid, expired or used up
C --> UI : code cannot be applied
UI --> G : show reason
end
deactivate C
deactivate UI
deactivate G
@enduml
