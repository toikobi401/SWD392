@startuml
title SD-01 UC-M09 Update Room Info

actor "Manager" as M
boundary "RoomInventoryPage" as UI
boundary "RoomController" as C
entity "Room" as R
database "MongoDB" as DB

M -> UI : edit room info and save
activate M
activate UI
UI -> C : submit room changes
activate C
C -> C : validate request
C -> R : update room info
activate R
R -> DB : save room
activate DB
DB --> R : save result
deactivate DB
R --> C : update result
deactivate R
alt room number is unique
C --> UI : room updated
UI --> M : show updated room details
else room number already exists
C --> UI : room number already exists
UI --> M : show error message
end
deactivate C
deactivate UI
deactivate M
@enduml
