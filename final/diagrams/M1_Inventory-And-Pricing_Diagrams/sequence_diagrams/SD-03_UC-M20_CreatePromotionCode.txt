@startuml
title SD-03 UC-M20 Create Promotion Code

actor "Manager" as M
boundary "PromotionsPage" as UI
boundary "PromotionController" as C
entity "Promotion" as P
entity "AuditLog" as A
entity "Booking" as B
database "MongoDB" as DB

M -> UI : setCreating(true)
activate M
activate UI
UI --> M : show new-code form and worked discount example\n10 percent, today through day 30, unlimited uses,\nno minimum spend, not public
deactivate UI
deactivate M

M -> UI : submit(e)
activate M
activate UI
UI -> C : POST /api/promotions (form)\ncreate(req, res)
activate C
C -> C : authenticate(req), requirePermission(MANAGE_PRICING)

alt missing, expired or invalid access token
    C --> UI : 401 UNAUTHORIZED / TOKEN_EXPIRED / INVALID_TOKEN
    UI --> M : show access error
else missing MANAGE_PRICING permission
    C --> UI : 403 FORBIDDEN
    UI --> M : show permission error
else authorized request
    C -> C : validateBody(rules), inputFrom(req.body),\ntermsFrom(input), problemsWith(terms)

    alt missing fields or invalid terms (UC-M11 E1)
        C --> UI : 400 VALIDATION_ERROR (problems)
        UI --> M : show problems and keep form for correction
    else valid terms
        C -> P : exists({code: terms.code})
        activate P
        P -> DB : query normalized code
        activate DB
        DB --> P : {_id} or null
        deactivate DB
        P --> C : {_id} or null
        deactivate P

        alt code already exists (UC-M11 E2)
            C --> UI : 409 PROMOTION_CODE_TAKEN
            UI --> M : show "Code already exists - choose another"
        else code not found
            C -> P : create({...terms, isPublic: Boolean(input.isPublic),\nisActive: true, usedCount: 0, createdBy: actorId})
            activate P
            P -> DB : insert promotion (unique code index)
            activate DB
            alt another Manager creates the same code first
                DB --> P : duplicate key error 11000
                P --> C : error 11000
                C --> UI : 409 PROMOTION_CODE_TAKEN
                UI --> M : show "Code already exists - choose another"
            else inserted
                DB --> P : created document
                P --> C : promotion
            end
            deactivate DB
            deactivate P

            opt promotion was created successfully
                C -> C : redact(after)
                C -> A : create({action: 'PROMOTION_CREATED', entityType: 'Promotion',\nentityId: String(promotion._id), actorId, after, timestamp: now})
                activate A
                A -> DB : insert append-only audit entry
                activate DB
                alt audit write succeeds
                    DB --> A : audit document
                    A --> C : audit entry
                else audit write fails
                    DB --> A : persistence error
                    A --> C : error (logged and swallowed; creation continues)
                end
                deactivate DB
                deactivate A

                C -> P : findById(id)
                activate P
                P -> DB : query newly created promotion by _id
                activate DB
                DB --> P : promotion document or null
                deactivate DB
                P --> C : promotion or null
                deactivate P

                alt promotion no longer exists
                    C --> UI : 404 PROMOTION_NOT_FOUND
                    UI --> M : show "Promotion code not found"
                else promotion found
                    C -> B : aggregate(pipeline)
                    activate B
                    B -> DB : match promotionId; group booking count, standing rooms,\nreservation codes, discount and revenue
                    activate DB
                    DB --> B : usage rows (normally empty for a new code)
                    deactivate DB
                    B --> C : usage rows
                    deactivate B
                    C -> C : view(promotion, usage, now)
                    C --> UI : 201 Created (PromotionView)
                    UI --> M : refresh promotions and open the new code
                end
            end
        end
    end
end

deactivate C
deactivate UI
deactivate M
@enduml
