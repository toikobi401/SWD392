activate UI

activate C
C -> C : validate request
C -> R : get room
activate R

activate DB

deactivate DB

deactivate R

activate R

activate DB

deactivate DB

deactivate R

UI --> RE : refresh room rack
else status change not allowed

end

deactivate UI
deactivate RE
