@startuml
title CD-02 UC-R13 Update Room Status

skinparam classAttributeIconSize 0
hide empty members

class RoomRackPage <<boundary>> {
    +changeStatus(roomId, status)
}

class Router <<boundary>> {
    +route(PATCH /api/rooms/:id/status)
    +authenticate(req)
    +requirePermission(permission)
}

class FrontDeskController <<boundary>> {
    +updateRoomStatus(req, res)
}

class RoomStatusService <<control>> {
    +apply(roomId, event, notes) : Room
}

class Room <<entity>> {
    - _id : ObjectId
    - roomNumber : String
    - status : RoomStatus
    - notes : String
    - version : Number
    +getRoom(id) : Room
    +updateStatus(id, status) : Room
}

enum RoomStatus <<enumeration>> {
    VACANT_CLEAN
    VACANT_DIRTY
    OCCUPIED
    INSPECTED
    OUT_OF_ORDER
}

RoomRackPage ..> Router : PATCH /api/rooms/:id/status
Router ..> FrontDeskController
FrontDeskController ..> RoomStatusService
RoomStatusService ..> Room
Room --> RoomStatus : status

@enduml
