class RoomInventoryPage <<boundary>> {
    +saveRoomInfo()
}

class Router <<boundary>> {
    +route(PATCH /api/rooms/:id)
    +authenticate(req)
    +requirePermission(permission)
    +validateBody(rules)
}

class RoomController <<boundary>> {
    +updateRoom(req, res)
}

class RoomInventoryService <<control>> {
    +updateRoomInfo(roomId, changes) : Room
}

class Room <<entity>> {
    - _id : ObjectId
    - roomNumber : String
    - floor : Number
    - notes : String
    - version : Number
    - status : RoomStatus
    +updateRoomInfo(id, data) : Room
}

enum RoomStatus <<enumeration>> {
    VACANT_CLEAN
    VACANT_DIRTY
    OCCUPIED
    INSPECTED
    OUT_OF_ORDER
}

RoomInventoryPage ..> Router : PATCH /api/rooms/:id
Router ..> RoomController
RoomController ..> RoomInventoryService
RoomInventoryService ..> Room
Room --> RoomStatus : status
