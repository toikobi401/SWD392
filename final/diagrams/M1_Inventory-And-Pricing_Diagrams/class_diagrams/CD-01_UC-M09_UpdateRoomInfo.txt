class RoomInventoryPage <<boundary>> {
    + selectRoomToEdit(id)
    + saveRoomInfo(roomNumber, floor, notes, version)
}

class RoomController <<boundary>> {
    + getRoom(req, res)
    + updateRoom(req, res)
    + authenticate(req)
    + requirePermission(permission)
    + validateInput(input)
}

class Room <<entity>> {
    - _id : ObjectId
    - roomNumber : String
    - floor : Number
    - notes : String
    - version : Number
    - status : RoomStatus
    + findById(id) : Room
    + findOneAndUpdate(filter, update) : Room
}

enum RoomStatus <<enumeration>> {
    VACANT_CLEAN
    VACANT_DIRTY
    OCCUPIED
    INSPECTED
    OUT_OF_ORDER
}

RoomInventoryPage ..> RoomController : GET /api/rooms/:id\nPATCH /api/rooms/:id
RoomController ..> Room
Room --> RoomStatus : status

note bottom of RoomInventoryPage
    SD-01 action mapping:
    "select a room to edit" = selectRoomToEdit(id)
    "edit roomNumber, floor, notes and save"
    = saveRoomInfo(roomNumber, floor, notes, version).
    Dashed responses return results to these actions.
end note

note bottom of RoomController
    Design only: RoomInventoryPage, getRoom and updateRoom
    and both routes are proposed by the unchanged SD-01.
    GET maps to getRoom; PATCH maps to updateRoom.
    "validate input" maps to validateInput.
    Middleware is folded into this boundary.
    PATCH uses _id + version, $set fields and $inc version.
    Returns 400, 401/403, 409 for duplicate roomNumber,
    409 ROOM_CHANGED for no match, or 200 with the room.
end note
@enduml
