@startuml
title CD-04 UC-G11 Apply Discount Code

skinparam classAttributeIconSize 0
hide empty members

class BookingWizardPage <<boundary>> {
    +applyDiscountCode(code)
}

class Router <<boundary>> {
    +route(GET /api/promotions/validate)
}

class PromotionController <<boundary>> {
    +validate(req, res)
}

class PromotionService <<control>> {
    +preview(code, subtotal) : PromotionPreview
}

class Promotion <<entity>> {
    - _id : ObjectId
    - code : String
    - description : String
    - discountType : DiscountType
    - discountValue : Number
    - validFrom : Date
    - validTo : Date
    - usageLimit : Number
    - usedCount : Number
    - minimumSpend : Number
    - isActive : Boolean
    +findByCode(code) : Promotion
}

enum DiscountType <<enumeration>> {
    PERCENTAGE
    FIXED_AMOUNT
}

BookingWizardPage ..> Router : GET /api/promotions/validate
Router ..> PromotionController
PromotionController ..> PromotionService
PromotionService ..> Promotion
Promotion --> DiscountType : discountType

@enduml
