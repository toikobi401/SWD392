class BookingWizardPage <<boundary>> {
    + applyCode(code)
}

class PromotionController <<boundary>> {
    + validate(req, res)
    + applicability(promotion, subtotal, now)
    + discountFor(promotion, subtotal) : Number
}

class Promotion <<entity>> {
    - _id : ObjectId
    - code : String
    - description : String
    - discountType : DiscountType
    - discountValue : Number
    - validFrom : Date
    - validTo : Date
    - usageLimit : Number
    - usedCount : Number
    - minimumSpend : Number
    - isActive : Boolean
    + findOne(filter) : Promotion
}

enum DiscountType <<enumeration>> {
    PERCENTAGE
    FIXED_AMOUNT
}

BookingWizardPage ..> PromotionController : GET /api/promotions/validate?code=&subtotal=
PromotionController ..> Promotion
Promotion --> DiscountType : discountType

note bottom of PromotionController
    Public preview: no authentication or database write.
    PromotionService.preview/resolve logic is folded in.
    applicability is PromotionRule.applicability;
    discountFor is PricingRule.discountFor.
    These helpers are not additional HTTP handlers.
    Validity uses booking time, not the stay dates.
    Booking creation rechecks and reprices the code (UC-G07).
end note
