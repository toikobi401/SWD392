@startuml
title CD-03 UC-M20 Create Promotion Code

skinparam classAttributeIconSize 0
hide empty members

class PromotionsPage <<boundary>> {
    + setCreating(value)
    + submit(e)
}

class PromotionController <<boundary>> {
    + create(req, res)
    + authenticate(req)
    + requirePermission(permission)
    + validateBody(rules)
    + inputFrom(body)
    + termsFrom(input)
    + problemsWith(terms) : String[]
    + redact(value)
    + view(promotion, usage, now)
}

class Promotion <<entity>> {
    - _id : ObjectId
    - code : String
    - description : String
    - discountType : DiscountType
    - discountValue : Number
    - validFrom : Date
    - validTo : Date
    - usageLimit : Number
    - usedCount : Number
    - minimumSpend : Number
    - isActive : Boolean
    - isPublic : Boolean
    - createdBy : ObjectId
    - createdAt : Date
    + exists(filter) : ObjectIdResult
    + create(data) : Promotion
    + findById(id) : Promotion
}

class AuditLog <<entity>> {
    - _id : ObjectId
    - actorId : ObjectId
    - action : String
    - entityType : String
    - entityId : String
    - after : Mixed
    - timestamp : Date
    + create(entry) : AuditLog
}

class Booking <<entity>> {
    - _id : ObjectId
    - promotionId : ObjectId
    - reservationCode : String
    - status : BookingStatus
    - discount : Number
    - totalAmount : Number
    + aggregate(pipeline) : UsageRow[]
}

enum DiscountType <<enumeration>> {
    PERCENTAGE
    FIXED_AMOUNT
}

enum BookingStatus <<enumeration>> {
    PENDING
    CONFIRMED
    CHECKED_IN
    CHECKED_OUT
    CANCELLED
    NO_SHOW
}

PromotionsPage ..> PromotionController : POST /api/promotions
PromotionController ..> Promotion
PromotionController ..> AuditLog
PromotionController ..> Booking
Promotion "0..1" -- "0..*" Booking : promotionId
Promotion --> DiscountType : discountType
Booking --> BookingStatus : status

note bottom of PromotionController
    Middleware, PromotionService, PromotionRule and
    AuditService helpers are folded into this boundary.
    Only create(req, res) is the HTTP handler.
    AuditLog.entityId is a String, not a schema reference.
    Booking usage is read by detail() after creation.
end note

note bottom of PromotionsPage
    submit is PromotionDialog's form handler,
    represented within its owning PromotionsPage.
    setCreating(true) opens the creation dialog.
end note
@enduml
