@startuml
title CD-03 UC-M20 Create Promotion Code

skinparam classAttributeIconSize 0
hide empty members

class PromotionsPage <<boundary>> {
    +createPromotion()
}

class Router <<boundary>> {
    +route(POST /api/promotions)
    +authenticate(req)
    +requirePermission(permission)
    +validateBody(rules)
}

class PromotionController <<boundary>> {
    +create(req, res)
}

class PromotionService <<control>> {
    +create(input, managerId) : PromotionView
}

class Promotion <<entity>> {
    - _id : ObjectId
    - code : String
    - description : String
    - discountType : DiscountType
    - discountValue : Number
    - validFrom : Date
    - validTo : Date
    - usageLimit : Number
    - usedCount : Number
    - minimumSpend : Number
    - isActive : Boolean
    - isPublic : Boolean
    - createdBy : ObjectId
    - createdAt : Date
    +codeExists(code) : Boolean
    +createPromotion(data) : Promotion
}

enum DiscountType <<enumeration>> {
    PERCENTAGE
    FIXED_AMOUNT
}

PromotionsPage ..> Router : POST /api/promotions
Router ..> PromotionController
PromotionController ..> PromotionService
PromotionService ..> Promotion
Promotion --> DiscountType : discountType

@enduml
