@startuml
title SD-15 UC-C17 Request Refund

actor "Customer" as U
boundary "BookingDetailPage" as UI
boundary "RefundController" as C
entity "RefundRequest" as R
database "MongoDB" as DB

U -> UI : request refund
activate U
activate UI
UI -> C : submit refund request
activate C
C -> C : validate request and refund eligibility
alt within 90-day refund window
C -> R : create refund request
activate R
R -> DB : save request
activate DB
DB --> R : request saved
deactivate DB
R --> C : refund request
deactivate R
C --> UI : refund request submitted
UI --> U : show reference and status
else refund window expired
C --> UI : refund not allowed
UI --> U : show error message
end
deactivate C
deactivate UI
deactivate U
@enduml
