@startuml
title SD-15 UC-C17 Request Refund

actor "Customer" as U
boundary "BookingDetailPage" as UI
boundary "RefundController" as C
entity "RefundRequest" as R
database "MongoDB" as DB
participant "Email service" as X

note over UI, R
    BookingDetailPage includes RefundDialog from CustomerPages.tsx.
    UI offers requests for CANCELLED bookings after seven days.
    Booking/payment eligibility queries and audit work are folded into C.
    No attachment, non-refundable-rate check or manager notification is implemented.
end note

U -> UI : setRefunding(true)
activate U
activate UI
UI --> U : show amount, reasonCategory and description form
deactivate UI
deactivate U

U -> UI : send.mutate()
activate U
activate UI
UI -> C : POST /api/refund-requests (bookingId, amount, reasonCategory, description)\nrequest(req, res)
activate C
C -> C : authenticate()
C -> C : validateBody(bookingId, amount >= 1, reasonCategory)
alt invalid or expired authentication
    C --> UI : 401 UNAUTHORIZED / INVALID_TOKEN / TOKEN_EXPIRED
    UI --> U : show authentication error
else invalid body
    C --> UI : 400 VALIDATION_ERROR
    UI --> U : show field errors
else valid authenticated request
    C -> C : request(req, res)\nBooking.findById(bookingId); check customerId
    alt booking missing
        C --> UI : 404 BOOKING_NOT_FOUND
        UI --> U : show booking not found
    else customer does not own booking
        C --> UI : 403 FORBIDDEN
        UI --> U : show ownership error
    else customer owns booking
        C -> C : request(req, res)\nPayment.findOne({bookingId, status:PAID}).sort({paidAt:-1}); check age
        alt no captured payment
            C --> UI : 400 NOT_REFUNDABLE
            UI --> U : show no captured payment exists
        else payment is older than 90 days
            C --> UI : 400 WINDOW_EXPIRED
            UI --> U : show refund window expired
        else payment eligible by age
            C -> R : findOne({paymentId, status:{$in:[PENDING, APPROVED]}})
            activate R
            R -> DB : query open request for payment
            activate DB
            DB --> R : open request or null
            deactivate DB
            R --> C : open request or null
            deactivate R
            alt open request exists
                C --> UI : 409 DUPLICATE_REQUEST (referenceNumber, status)
                UI --> U : show error with existing reference
            else no open request
                C -> C : request(req, res)\nPayment.find({reversalOfId, status:{$in:[REVERSED, FAILED]}})\nremaining = payment.amount - committed reversals
                alt remaining <= 0
                    C --> UI : 409 ALREADY_REFUNDED
                    UI --> U : show payment already refunded in full
                else amount remains refundable
                    C -> C : request(req, res)\namount = min(requested amount, remaining); generateReference()
                    C -> R : create({referenceNumber, bookingId, paymentId, requestedBy,\nrequestedAmount:amount, reasonCategory, description, status:PENDING})
                    activate R
                    R -> DB : insert refund request
                    activate DB
                    DB --> R : saved request
                    deactivate DB
                    R --> C : request
                    deactivate R
                    C -> C : record(REFUND_REQUESTED)
                    alt amount > 500000 (manager review)
                        C -> X : sendDecisionNotice(guest.email, acknowledgement, referenceNumber)
                        activate X
                        X --> C : delivered or delivery error (caught)
                        deactivate X
                        C --> UI : 201 Created (referenceNumber, status:PENDING,\nrequestedAmount, approvedAmount)
                        UI --> U : close dialog; show reference and reply within 3 working days
                        ref over C, R : Manager approval, partial approval or rejection: SD-16.
                    else amount <= 500000 (automatic approval attempted)
                        C -> C : decide({refundId, approverId:"SYSTEM", approve:true, approvedAmount:amount})
                        C -> R : findById(refundId)
                        activate R
                        R -> DB : read pending request
                        activate DB
                        DB --> R : request
                        deactivate DB
                        R --> C : request
                        deactivate R
                        C -> R : save()
                        activate R
                        note over C, R
                            Attempt APPROVED, approvedAmount, decidedAt, decisionNote,
                            decidedBy="SYSTEM". Schema requires decidedBy:ObjectId.
                            The TypeScript "as never" cast does not convert the value.
                        end note
                        R --> C : ValidationError before database write
                        deactivate R
                        C --> UI : 400 VALIDATION_ERROR
                        UI --> U : show validation error; dialog remains open
                        note over R, DB
                            Current-code inference: created request remains PENDING.
                            Automatic refund settlement/email is not reached.
                            Retrying encounters DUPLICATE_REQUEST.
                        end note
                    end
                end
            end
        end
    end
end
deactivate C
deactivate UI
deactivate U
@enduml
