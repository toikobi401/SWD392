@startuml
title SD-13 UC-G10 Pay for Booking

actor "Guest" as G
boundary "PaymentResultPage" as UI
boundary "PaymentController" as C
entity "Payment" as P
database "MongoDB" as DB
participant "payOS" as X

G -> X : pay on payOS checkout
activate G
activate X
X -> C : payment result webhook
activate C
C -> C : validate request
C -> P : update payment status
activate P
P -> DB : save payment
activate DB
DB --> P : payment saved
deactivate DB
P --> C : payment
deactivate P
C --> X : webhook received
X --> UI : redirect to result page
deactivate X
UI -> C : get payment result
activate UI
C -> P : get payment
activate P
P -> DB : find payment
activate DB
DB --> P : payment data
deactivate DB
P --> C : payment
deactivate P
alt payment is PAID
C --> UI : payment confirmed
UI --> G : show booking confirmation
else payment failed or cancelled
C --> UI : payment not completed
UI --> G : show failure and retry option
end
deactivate UI
deactivate C
deactivate G
@enduml
