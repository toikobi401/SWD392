@startuml
title SD-14 UC-R15 Process Payment

actor "Receptionist" as R
boundary "FolioPage" as UI
boundary "FrontDeskController" as C
entity "Folio" as F
entity "Payment" as P
database "MongoDB" as DB

R -> UI : enter amount and payment method
activate R
activate UI
UI -> C : submit payment
activate C
C -> C : validate request
C -> F : get folio
activate F
F -> DB : find folio
activate DB
DB --> F : folio data
deactivate DB
F --> C : folio
deactivate F
alt folio open and amount within balance
C -> P : record payment
activate P
P -> DB : save payment
activate DB
DB --> P : payment saved
deactivate DB
P --> C : payment
deactivate P
C -> F : update folio balance
activate F
F -> DB : save folio
activate DB
DB --> F : folio saved
deactivate DB
F --> C : updated folio
deactivate F
C --> UI : payment recorded
UI --> R : show receipt and new balance
else amount exceeds balance
C --> UI : payment rejected
UI --> R : show error message
end
deactivate C
deactivate UI
deactivate R
@enduml
