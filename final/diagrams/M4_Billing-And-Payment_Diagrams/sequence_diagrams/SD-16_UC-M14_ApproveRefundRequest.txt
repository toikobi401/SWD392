actor "Manager" as M
boundary "ApprovalsPage" as UI
boundary "RefundController" as C
entity "RefundRequest" as R
database "MongoDB" as DB

M -> UI : open pending refunds
activate M
activate UI
UI -> C : request pending refunds
activate C
C -> R : get pending refund requests
activate R
R -> DB : find pending requests
activate DB
DB --> R : requests
deactivate DB
R --> C : pending requests
deactivate R
C --> UI : pending refund list
UI --> M : show pending list
M -> UI : approve request with amount
UI -> C : submit decision
C -> C : validate request
alt amount within captured total
C -> R : approve refund request
activate R
R -> DB : save decision
activate DB
DB --> R : decision saved
deactivate DB
R --> C : updated request
deactivate R
C --> UI : refund approved
UI --> M : remove request from list
else amount exceeds captured total
C --> UI : invalid refund amount
UI --> M : show error message
end
deactivate C
deactivate UI
deactivate M
