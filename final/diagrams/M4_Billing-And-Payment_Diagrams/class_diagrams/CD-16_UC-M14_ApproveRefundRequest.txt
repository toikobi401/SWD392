@startuml
title CD-16 UC-M14 Approve Refund Request

skinparam classAttributeIconSize 0
hide empty members

class ApprovalsPage <<boundary>> {
    +loadPendingRefunds()
    +approveRefund(id, amount)
}

class RefundController <<boundary>> {
    +pending(req, res)
    +decide(req, res)
    +validateRequest(req)
}

class RefundRequest <<entity>> {
    - _id : ObjectId
    - referenceNumber : String
    - bookingId : ObjectId
    - paymentId : ObjectId
    - requestedBy : ObjectId
    - requestedAmount : Number
    - approvedAmount : Number
    - reasonCategory : String
    - description : String
    - status : RefundStatus
    - decidedBy : ObjectId
    - decidedAt : Date
    - decisionNote : String
    - createdAt : Date
    +getPending() : RefundRequest[]
    +approve(id, amount) : RefundRequest
}

enum RefundStatus <<enumeration>> {
    PENDING
    APPROVED
    REJECTED
    COMPLETED
    REFUND_FAILED
}

ApprovalsPage ..> RefundController : GET /api/refund-requests/pending
ApprovalsPage ..> RefundController : PATCH /api/refund-requests/:id/decision
RefundController ..> RefundRequest
RefundRequest --> RefundStatus

@enduml
