@startuml
title CD-16 UC-M14 Approve Refund Request

skinparam classAttributeIconSize 0
hide empty members

class ApprovalsPage <<boundary>> {
    +loadPendingRefunds()
    +approveRefund(id, amount)
}

class Router <<boundary>> {
    +route(GET /api/refund-requests/pending)
    +route(PATCH /api/refund-requests/:id/decision)
    +authenticate(req)
    +requirePermission(permission)
}

class RefundController <<boundary>> {
    +pending(req, res)
    +decide(req, res)
}

class RefundService <<control>> {
    +pendingQueue() : RefundRequest[]
    +decide(command) : RefundRequest
}

class RefundRequest <<entity>> {
    - _id : ObjectId
    - referenceNumber : String
    - bookingId : ObjectId
    - paymentId : ObjectId
    - requestedBy : ObjectId
    - requestedAmount : Number
    - approvedAmount : Number
    - reasonCategory : String
    - description : String
    - status : RefundStatus
    - decidedBy : ObjectId
    - decidedAt : Date
    - decisionNote : String
    - createdAt : Date
    +getPending() : RefundRequest[]
    +approve(id, amount) : RefundRequest
}

enum RefundStatus <<enumeration>> {
    PENDING
    APPROVED
    REJECTED
    COMPLETED
    REFUND_FAILED
}

ApprovalsPage ..> Router : GET /api/refund-requests/pending\nPATCH /api/refund-requests/:id/decision
Router ..> RefundController
RefundController ..> RefundService
RefundService ..> RefundRequest
RefundRequest --> RefundStatus

@enduml
