    - referenceNumber : String

    - requestedBy : ObjectId
    - requestedAmount : Number
    - approvedAmount : Number
    - reasonCategory : String
    - description : String
    - status : RefundStatus
    - decidedBy : ObjectId
    - decidedAt : Date
    - decisionNote : String
    - createdAt : Date
    + find(filter) : Query<RefundRequest[]>
    + findById(id) : RefundRequest
    + save() : RefundRequest
}

class EmailService <<external>> {
    + sendDecisionNotice(to, subject, body)
}

enum RefundStatus <<enumeration>> {
    PENDING
    APPROVED
    REJECTED
    COMPLETED
    REFUND_FAILED
}
