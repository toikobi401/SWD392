@startuml
title CD-13 UC-G10 Pay for Booking

skinparam classAttributeIconSize 0
hide empty members

class PaymentResultPage <<boundary>> {
    +loadResult(orderCode)
}

class PaymentController <<boundary>> {
    +webhook(req, res)
    +reconcile(req, res)
    +validateRequest(req)
}

class Payment <<entity>> {
    - _id : ObjectId
    - bookingId : ObjectId
    - amount : Number
    - method : PaymentMethod
    - status : PaymentStatus
    - idempotencyKey : String
    - gatewayRef : String
    - gatewayMessage : String
    - gatewayOrderCode : Number
    - checkoutUrl : String
    - qrCode : String
    - expiresAt : Date
    - notifiedAt : Date
    - paidAt : Date
    +updateStatus(orderCode, status) : Payment
    +getPayment(orderCode) : Payment
}

class PayOS <<external>> {
    +checkout(order)
    +sendWebhook(result)
}

enum PaymentStatus <<enumeration>> {
    PENDING
    PAID
    FAILED
    REVERSED
}

enum PaymentMethod <<enumeration>> {
    CARD
    CASH
    BANK_TRANSFER
    E_WALLET
    VOUCHER
    LOYALTY_POINTS
}

PaymentResultPage ..> PaymentController : GET /api/payments/payos/:orderCode/reconcile
PaymentController ..> Payment
PaymentController ..> PayOS
PayOS ..> PaymentController : POST /api/payments/payos/webhook
Payment --> PaymentStatus
Payment --> PaymentMethod

@enduml
