@startuml
title CD-13 UC-G10 Pay for Booking

skinparam classAttributeIconSize 0
hide empty members

class PaymentResultPage <<boundary>> {
    +loadResult(orderCode)
}

class Router <<boundary>> {
    +route(POST /api/payments/payos/webhook)
    +route(GET /api/payments/payos/:orderCode/reconcile)
}

class PaymentController <<boundary>> {
    +webhook(req, res)
    +reconcile(req, res)
}

class PaymentService <<control>> {
    +handleWebhook(body)
    +reconcile(orderCode) : PaymentStatus
}

class Payment <<entity>> {
    - _id : ObjectId
    - bookingId : ObjectId
    - amount : Number
    - method : PaymentMethod
    - status : PaymentStatus
    - idempotencyKey : String
    - gatewayRef : String
    - gatewayMessage : String
    - gatewayOrderCode : Number
    - checkoutUrl : String
    - qrCode : String
    - expiresAt : Date
    - notifiedAt : Date
    - paidAt : Date
    +updateStatus(orderCode, status) : Payment
    +getPayment(orderCode) : Payment
}

class PayOS <<external>> {
    +checkout(order)
    +sendWebhook(result)
}

enum PaymentStatus <<enumeration>> {
    PENDING
    PAID
    FAILED
    REVERSED
}

enum PaymentMethod <<enumeration>> {
    CARD
    CASH
    BANK_TRANSFER
    E_WALLET
    VOUCHER
    LOYALTY_POINTS
}

PayOS ..> Router : POST /api/payments/payos/webhook
PaymentResultPage ..> Router : GET /api/payments/payos/:orderCode/reconcile
Router ..> PaymentController
PaymentController ..> PaymentService
PaymentService ..> Payment
Payment --> PaymentStatus
Payment --> PaymentMethod

@enduml
