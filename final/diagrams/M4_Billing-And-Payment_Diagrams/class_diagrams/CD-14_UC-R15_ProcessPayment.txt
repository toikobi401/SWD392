class FolioPage <<boundary>> {
    +takePayment(amount, method)
}

class FrontDeskController <<boundary>> {
    +takePayment(req, res)
    +validateRequest(req)
}

class Folio <<entity>> {
    - _id : ObjectId
    - bookingId : ObjectId
    - status : FolioStatus
    - totalPayments : Number
    - balance : Number
    +getFolio(id) : Folio
    +updateBalance(id, amount) : Folio
}

class Payment <<entity>> {
    - _id : ObjectId
    - bookingId : ObjectId
    - folioId : ObjectId
    - amount : Number
    - method : PaymentMethod
    - status : PaymentStatus
    - idempotencyKey : String
    - gatewayRef : String
    - gatewayMessage : String
    - gatewayOrderCode : Number
    - checkoutUrl : String
    - qrCode : String
    - cashierId : ObjectId
    - drawerSessionId : ObjectId
    - paidAt : Date
    +recordPayment(data) : Payment
}

enum FolioStatus <<enumeration>> {
    OPEN
    CLOSED
}

enum PaymentStatus <<enumeration>> {
    PENDING
    PAID
    FAILED
    REVERSED
}

enum PaymentMethod <<enumeration>> {
    CARD
    CASH
    BANK_TRANSFER
    E_WALLET
    VOUCHER
    LOYALTY_POINTS
}

FolioPage ..> FrontDeskController : POST /api/folios/:id/payments
FrontDeskController ..> Folio
FrontDeskController ..> Payment
Folio "0..1" -- "0..*" Payment : folioId
Folio --> FolioStatus
Payment --> PaymentStatus
Payment --> PaymentMethod

@enduml
