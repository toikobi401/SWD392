@startuml
title CD-15 UC-C17 Request Refund

skinparam classAttributeIconSize 0
hide empty members

class BookingDetailPage <<boundary>> {
    +requestRefund(bookingId, amount, reason)
}

class Router <<boundary>> {
    +route(POST /api/refund-requests)
    +authenticate(req)
    +validateBody(rules)
}

class RefundController <<boundary>> {
    +request(req, res)
}

class RefundService <<control>> {
    +request(command) : RefundRequest
    +checkEligibility(booking, payment)
}

class RefundRequest <<entity>> {
    - _id : ObjectId
    - referenceNumber : String
    - bookingId : ObjectId
    - paymentId : ObjectId
    - requestedBy : ObjectId
    - requestedAmount : Number
    - approvedAmount : Number
    - reasonCategory : String
    - description : String
    - status : RefundStatus
    - decidedBy : ObjectId
    - decidedAt : Date
    - decisionNote : String
    +createRequest(data) : RefundRequest
}

enum RefundStatus <<enumeration>> {
    PENDING
    APPROVED
    REJECTED
    COMPLETED
    REFUND_FAILED
}

BookingDetailPage ..> Router : POST /api/refund-requests
Router ..> RefundController
RefundController ..> RefundService
RefundService ..> RefundRequest
RefundRequest --> RefundStatus

@enduml
