    +validateRequest(req)
}

class RefundRequest <<entity>> {
    - _id : ObjectId
    - referenceNumber : String
    - bookingId : ObjectId
    - paymentId : ObjectId
    - requestedBy : ObjectId
    - requestedAmount : Number
    - approvedAmount : Number
    - reasonCategory : String
    - description : String
    - status : RefundStatus
    - decidedBy : ObjectId
    - decidedAt : Date
    - decisionNote : String
    +createRequest(data) : RefundRequest
}

enum RefundStatus <<enumeration>> {
    PENDING
    APPROVED
    REJECTED
    COMPLETED
    REFUND_FAILED
}
