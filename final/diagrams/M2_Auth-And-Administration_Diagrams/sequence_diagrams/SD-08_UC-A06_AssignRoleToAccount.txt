@startuml
title SD-08 UC-A06 Assign Role to Account

actor "Admin" as A
boundary "AccountsPage" as UI
boundary "AccountController" as C
entity "User" as U
entity "Role" as R
database "MongoDB" as DB

A -> UI : choose account and role
activate A
activate UI
UI -> C : submit role assignment
activate C
C -> C : validate request
C -> U : get account
activate U
U -> DB : find user
activate DB
DB --> U : user data
deactivate DB
U --> C : account
deactivate U
C -> R : get role
activate R
R -> DB : find role
activate DB
DB --> R : role data
deactivate DB
R --> C : role
deactivate R
alt role can be assigned
C -> U : add role to account
activate U
U -> DB : save user
activate DB
DB --> U : user saved
deactivate DB
U --> C : updated account
deactivate U
C --> UI : role assigned
UI --> A : show updated roles
else role conflicts with current roles
C --> UI : role conflict
UI --> A : show error message
end
deactivate C
deactivate UI
deactivate A
@enduml
