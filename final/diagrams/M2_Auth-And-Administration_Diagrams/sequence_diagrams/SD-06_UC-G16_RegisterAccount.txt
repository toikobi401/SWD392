@startuml
title SD-06 UC-G16 Register Account

actor "Guest" as G
boundary "RegisterPage" as UI
boundary "Router" as RO
boundary "AuthController" as C
control "AuthService" as S
entity "User" as U
database "MongoDB" as DB
participant "Email service" as E

G -> UI : fill registration form
activate G
activate UI
UI -> RO : POST /api/auth/register
activate RO
RO -> RO : validate input
RO -> C : forward request
activate C
C -> S : register account
activate S
S -> U : check email exists
activate U
U -> DB : find user by email
activate DB
DB --> U : search result
deactivate DB
U --> S : email status
deactivate U
alt email not registered
S -> U : create account
activate U
U -> DB : save user
activate DB
DB --> U : user saved
deactivate DB
U --> S : new account
deactivate U
S -> E : send verification email
S --> C : new account
C --> UI : registration successful
UI --> G : ask to verify email
else email already registered
S --> C : email already registered
C --> UI : email already registered
UI --> G : show error message
end
deactivate S
deactivate C
deactivate RO
deactivate UI
deactivate G
@enduml
