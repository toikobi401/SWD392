@startuml
title SD-06 UC-G16 Register Account

actor "Guest" as G
boundary "RegisterPage" as UI
boundary "AuthController" as C
entity "User" as U
database "MongoDB" as DB
participant "Email service" as E

G -> UI : fill registration form
activate G
activate UI
UI -> C : submit registration
activate C
C -> C : validate request
C -> U : check email exists
activate U
U -> DB : find user by email
activate DB
DB --> U : search result
deactivate DB
U --> C : email status
deactivate U
alt email not registered
C -> U : create account
activate U
U -> DB : save user
activate DB
DB --> U : user saved
deactivate DB
U --> C : new account
deactivate U
C -> E : send verification email
C --> UI : registration successful
UI --> G : ask to verify email
else email already registered
C --> UI : email already registered
UI --> G : show error message
end
deactivate C
deactivate UI
deactivate G
@enduml
