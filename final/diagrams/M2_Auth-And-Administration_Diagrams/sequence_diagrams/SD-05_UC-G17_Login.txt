@startuml

boundary "LoginPage" as UI
boundary "AuthController" as C
entity "User" as U
database "MongoDB" as DB

G -> UI : enter email and password
activate G
activate UI
UI -> C : submit login
activate C
C -> C : validate request
C -> U : get account by email
activate U
U -> DB : find user
activate DB
DB --> U : user data
deactivate DB
U --> C : account
deactivate U
alt account active and password correct
C --> UI : access token
UI --> G : open home page
else wrong email or password
C --> UI : invalid credentials
UI --> G : show login error
end
deactivate C
deactivate UI
deactivate G
@enduml
