@startuml
title SD-07 UC-C03 Reset Forgotten Password

actor "Customer" as CU
boundary "ForgotPasswordPage" as UI
boundary "AuthController" as C
entity "User" as U
database "MongoDB" as DB
participant "Email service" as E


CU -> UI : request(e) - enter email and request code
activate CU
activate UI
UI -> C : POST /api/auth/forgot-password (email, channel: EMAIL)\nforgotPassword(req, res)
activate C
C -> U : findOne({ email })
activate U
U -> DB : query normalized email
activate DB
DB --> U : user document or null
deactivate DB
U --> C : user or null
deactivate U
alt account not found
    C --> UI : 200 OK (generic message)
    UI --> CU : "If that account exists, a verification code has been sent"
else account found
    C -> U : save()
    activate U
    U -> DB : save schema-backed user fields
    activate DB
    DB --> U : saved document (OTP fields are not persisted)
    deactivate DB
    U --> C : user
    deactivate U
    C -> E : sendOtp(email, otp, EMAIL)
    activate E
    alt provider accepts email
        E --> C : accepted
        C --> UI : 200 OK (generic message)
        UI --> CU : show code and new-password form
    else provider fails or is unconfigured in production
        E --> C : delivery error
        C --> UI : 500 INTERNAL_ERROR
        UI --> CU : show error; allow another request
    end
    deactivate E
end
deactivate C
deactivate UI
deactivate CU

opt Customer submits the reset form
    CU -> UI : reset(e) - enter email, otp and new password
    activate CU
    activate UI
    UI -> C : POST /api/auth/reset-password (email, otp, newPassword)\nresetPassword(req, res)
    activate C
    C -> C : assertPasswordStrength(newPassword)
    alt fewer than 8 characters or missing uppercase, lowercase or digit
        C --> UI : 400 WEAK_PASSWORD
        UI --> CU : show failed password strength rules
    else strong password
        C -> U : findOne({ email }).select('+passwordHash')
        activate U
        U -> DB : query normalized email and include passwordHash
        activate DB
        DB --> U : user document or null
        deactivate DB
        U --> C : user or null
        deactivate U
        alt no user, missing OTP fields, expired code or mismatched code
            C --> UI : 400 INVALID_OTP
            UI --> CU : "The code is invalid or has expired"
        else valid OTP fields available (intended flow; schema gap above)
            opt status is LOCKED
            end
            C -> U : save()
            activate U
            U -> DB : persist passwordHash, failedLoginAttempts and status
            activate DB
            DB --> U : saved document
            deactivate DB
            U --> C : saved user
            deactivate U
            C -> C : record(PASSWORD_RESET)
            C --> UI : 200 OK (message: Your password has been reset)
            UI --> CU : show success; navigate to /login after 1500 ms
        end
    end
    deactivate C
    deactivate UI
    deactivate CU
end

@enduml
