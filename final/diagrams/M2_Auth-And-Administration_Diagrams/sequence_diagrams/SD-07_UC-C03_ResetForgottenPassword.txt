@startuml
title SD-07 UC-C03 Reset Forgotten Password

actor "Customer" as CU
boundary "ForgotPasswordPage" as UI
boundary "Router" as RO
boundary "AuthController" as C
control "AuthService" as S
entity "User" as U
database "MongoDB" as DB
participant "Email service" as E

CU -> UI : enter email
activate CU
activate UI
UI -> RO : POST /api/auth/forgot-password
activate RO
RO -> C : forward request
activate C
C -> S : request reset code
activate S
S -> U : save reset code for email
activate U
U -> DB : update user
activate DB
DB --> U : saved
deactivate DB
U --> S : done
deactivate U
S -> E : send reset code email
S --> C : code sent
deactivate S
C --> UI : reset code sent
UI --> CU : ask for code and new password
CU -> UI : enter code and new password
UI -> RO : POST /api/auth/reset-password
RO -> C : forward request
C -> S : reset password
activate S
S -> U : reset password with code
activate U
U -> DB : update password
activate DB
DB --> U : update result
deactivate DB
U --> S : result
deactivate U
alt code is valid
S --> C : password changed
C --> UI : password changed
UI --> CU : go to login page
else code is wrong or expired
S --> C : invalid reset code
C --> UI : invalid reset code
UI --> CU : show error message
end
deactivate S
deactivate C
deactivate RO
deactivate UI
deactivate CU
@enduml
