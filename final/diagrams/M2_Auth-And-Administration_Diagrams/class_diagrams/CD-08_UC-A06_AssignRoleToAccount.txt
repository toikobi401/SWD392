@startuml
title CD-08 UC-A06 Assign Role to Account

skinparam classAttributeIconSize 0
hide empty members

class AccountsPage <<boundary>> {
    +assignRole(accountId, roleId)
}

class Router <<boundary>> {
    +route(POST /api/accounts/:id/roles)
    +authenticate(req)
    +requirePermission(permission)
}

class AccountController <<boundary>> {
    +assignRole(req, res)
}

class RoleAssignmentService <<control>> {
    +assign(command) : User
}

class User <<entity>> {
    - _id : ObjectId
    - email : String
    - status : AccountStatus
    - roles : ObjectId[]
    +getAccount(id) : User
    +addRole(id, roleId) : User
}

class Role <<entity>> {
    - _id : ObjectId
    - name : RoleName
    - permissions : Permission[]
    +getRole(id) : Role
}

enum AccountStatus <<enumeration>> {
    PENDING_VERIFICATION
    ACTIVE
    LOCKED
    DEACTIVATED
}

enum RoleName <<enumeration>> {
    GUEST
    CUSTOMER
    EMPLOYEE
    RECEPTIONIST
    MANAGER
    ADMIN
    SUPER_ADMIN
}

enum Permission <<enumeration>> {
    BOOKING_READ
    BOOKING_CREATE
    BOOKING_MODIFY
    BOOKING_CANCEL
    CHECK_IN
    CHECK_OUT
    ROOM_STATUS_UPDATE
    PROCESS_PAYMENT
    PROCESS_REFUND
    APPROVE_LEAVE
    APPROVE_REFUND
    APPROVE_PAYROLL
    MANAGE_INVENTORY
    MANAGE_PRICING
    VIEW_REPORTS
    MANAGE_ACCOUNTS
    MANAGE_ROLES
    MANAGE_SETTINGS
    VIEW_AUDIT_LOG
    MANAGE_BACKUP
}

AccountsPage ..> Router : POST /api/accounts/:id/roles
Router ..> AccountController
AccountController ..> RoleAssignmentService
RoleAssignmentService ..> User
RoleAssignmentService ..> Role
User "0..*" -- "0..*" Role : roles
User --> AccountStatus : status
Role --> RoleName : name
Role --> Permission : permissions

@enduml
