class ForgotPasswordPage <<boundary>> {
    + request(e)
    + reset(e)
}

class AuthController <<boundary>> {
    + forgotPassword(req, res)
    + resetPassword(req, res)
    + assertPasswordStrength(password)
    + record(entry)
}

class User <<entity>> {
    - _id : ObjectId
    - email : String
    - passwordHash : String
    - failedLoginAttempts : Number
    - status : AccountStatus
    + findOne(filter) : Query<User>
    + select(fields) : Query<User>
    + save() : Promise<User>
}

class EmailService <<external>> {
    + sendOtp(to, otp, channel)
}

enum AccountStatus <<enumeration>> {
    PENDING_VERIFICATION
    ACTIVE
    LOCKED
    DEACTIVATED
}

ForgotPasswordPage ..> AuthController : POST /api/auth/forgot-password\nPOST /api/auth/reset-password
AuthController ..> User
AuthController ..> EmailService : EMAIL channel
User --> AccountStatus : status
