@startuml
title CD-07 UC-C03 Reset Forgotten Password

skinparam classAttributeIconSize 0
hide empty members

class ForgotPasswordPage <<boundary>> {
    +requestCode(email)
    +resetPassword(code, newPassword)
}

class Router <<boundary>> {
    +route(POST /api/auth/forgot-password)
    +route(POST /api/auth/reset-password)
}

class AuthController <<boundary>> {
    +forgotPassword(req, res)
    +resetPassword(req, res)
}

class AuthService <<control>> {
    +requestPasswordReset(email, channel)
    +resetPassword(email, otp, newPassword)
}

class User <<entity>> {
    - _id : ObjectId
    - email : String
    - passwordHash : String
    - failedLoginAttempts : Number
    - status : AccountStatus
    +saveResetCode(email, code)
    +resetPassword(email, code, password) : Boolean
}

class EmailService <<external>> {
    +sendResetCode(to, code)
}

enum AccountStatus <<enumeration>> {
    PENDING_VERIFICATION
    ACTIVE
    LOCKED
    DEACTIVATED
}

ForgotPasswordPage ..> Router : POST /api/auth/forgot-password\nPOST /api/auth/reset-password
Router ..> AuthController
AuthController ..> AuthService
AuthService ..> User
AuthService ..> EmailService
User --> AccountStatus : status

@enduml
