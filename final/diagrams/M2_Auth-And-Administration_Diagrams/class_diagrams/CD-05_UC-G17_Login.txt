@startuml
title CD-05 UC-G17 Login

skinparam classAttributeIconSize 0
hide empty members

class LoginPage <<boundary>> {
    +login(email, password)
}

class Router <<boundary>> {
    +route(POST /api/auth/login)
    +validateBody(rules)
}

class AuthController <<boundary>> {
    +login(req, res)
}

class AuthService <<control>> {
    +login(command) : AuthTokens
}

class User <<entity>> {
    - _id : ObjectId
    - email : String
    - fullName : String
    - passwordHash : String
    - status : AccountStatus
    - roles : ObjectId[]
    - failedLoginAttempts : Number
    - lastLoginAt : Date
    +findByEmail(email) : User
}

enum AccountStatus <<enumeration>> {
    PENDING_VERIFICATION
    ACTIVE
    LOCKED
    DEACTIVATED
}

LoginPage ..> Router : POST /api/auth/login
Router ..> AuthController
AuthController ..> AuthService
AuthService ..> User
User --> AccountStatus : status

@enduml
