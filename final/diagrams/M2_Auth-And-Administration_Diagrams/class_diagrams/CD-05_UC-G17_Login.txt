@startuml

skinparam classAttributeIconSize 0
hide empty members

class LoginPage <<boundary>> {
    +login(email, password)
}

class AuthController <<boundary>> {
    +login(req, res)
    +validateRequest(req)
}

class User <<entity>> {
    - _id : ObjectId
    - email : String
    - fullName : String
    - passwordHash : String
    - status : AccountStatus
    - roles : ObjectId[]
    - failedLoginAttempts : Number
    - lastLoginAt : Date
    +findByEmail(email) : User
}

enum AccountStatus <<enumeration>> {
    PENDING_VERIFICATION
    ACTIVE
    LOCKED
    DEACTIVATED
}

LoginPage ..> AuthController : POST /api/auth/login
AuthController ..> User
User --> AccountStatus : status

@enduml
