@startuml
title CD-06 UC-G16 Register Account

skinparam classAttributeIconSize 0
hide empty members

class RegisterPage <<boundary>> {
    +register(form)
}

class Router <<boundary>> {
    +route(POST /api/auth/register)
    +validateBody(rules)
}

class AuthController <<boundary>> {
    +register(req, res)
}

class AuthService <<control>> {
    +register(command) : User
}

class User <<entity>> {
    - _id : ObjectId
    - fullName : String
    - email : String
    - phone : String
    - passwordHash : String
    - status : AccountStatus
    - roles : ObjectId[]
    - emailVerifiedAt : Date
    - userType : String
    +emailExists(email) : Boolean
    +createAccount(data) : User
}

class EmailService <<external>> {
    +sendVerificationEmail(to, link)
}

enum AccountStatus <<enumeration>> {
    PENDING_VERIFICATION
    ACTIVE
    LOCKED
    DEACTIVATED
}

RegisterPage ..> Router : POST /api/auth/register
Router ..> AuthController
AuthController ..> AuthService
AuthService ..> User
AuthService ..> EmailService
User --> AccountStatus : status

@enduml
