@startuml
title SD-17 UC-R06 Check In Guest

actor "Receptionist" as A
boundary "StaffBookingPage" as UI
boundary "FrontDeskController" as C
entity "Booking" as B
entity "Room" as R
database "MongoDB" as DB

A -> UI : verify guest ID and select room
activate A
activate UI
UI -> C : submit check-in
activate C
C -> C : validate request
C -> B : get booking
activate B
B -> DB : find booking
activate DB
DB --> B : booking data
deactivate DB
B --> C : booking
deactivate B
alt identity matches the booking
C -> R : assign room
activate R
R -> DB : save room as OCCUPIED
activate DB
DB --> R : room saved
deactivate DB
R --> C : assigned room
deactivate R
C -> B : mark booking checked in
activate B
B -> DB : save booking
activate DB
DB --> B : booking saved
deactivate DB
B --> C : updated booking
deactivate B
C --> UI : check-in completed
UI --> A : show room and stay details
else identity does not match the booking
C --> UI : identity check failed
UI --> A : show error message
end
deactivate C
deactivate UI
deactivate A
@enduml
