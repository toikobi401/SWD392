actor "Receptionist" as A
boundary "StaffBookingPage" as UI
boundary "FrontDeskController" as C
entity "Folio" as F
entity "Room" as R
database "MongoDB" as DB

A -> UI : request check-out
activate A
activate UI
UI -> C : submit check-out
activate C
C -> C : validate request
C -> F : get folio
activate F
F -> DB : find folio
activate DB
DB --> F : folio data
deactivate DB
F --> C : folio
deactivate F
alt balance is settled
C -> F : close folio
activate F
F -> DB : save folio
activate DB
DB --> F : folio saved
deactivate DB
F --> C : closed folio
deactivate F
C -> R : release room
activate R
R -> DB : save room as VACANT_DIRTY
activate DB
DB --> R : room saved
deactivate DB
R --> C : released room
deactivate R
C --> UI : check-out completed with invoice
UI --> A : show invoice
else outstanding balance
C --> UI : balance must be settled first
UI --> A : show balance due
end
deactivate C
deactivate UI
deactivate A
