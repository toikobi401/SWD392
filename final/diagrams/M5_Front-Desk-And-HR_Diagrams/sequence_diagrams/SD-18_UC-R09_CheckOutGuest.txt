@startuml
title SD-18 UC-R09 Check Out Guest

actor "Receptionist" as A
boundary "StaffBookingPage" as UI
boundary "Router" as RO
boundary "FrontDeskController" as C
control "CheckOutService" as S
entity "Folio" as F
entity "Room" as R
database "MongoDB" as DB

A -> UI : request check-out
activate A
activate UI
UI -> RO : POST /api/front-desk/check-out
activate RO
RO -> RO : authenticate, check permission, validate input
RO -> C : forward request
activate C
C -> S : check out guest
activate S
S -> F : get folio
activate F
F -> DB : find folio
activate DB
DB --> F : folio data
deactivate DB
F --> S : folio
deactivate F
alt balance is settled
S -> F : close folio
activate F
F -> DB : save folio
activate DB
DB --> F : folio saved
deactivate DB
F --> S : closed folio
deactivate F
S -> R : release room
activate R
R -> DB : save room as VACANT_DIRTY
activate DB
DB --> R : room saved
deactivate DB
R --> S : released room
deactivate R
S --> C : invoice
C --> UI : check-out completed with invoice
UI --> A : show invoice
else outstanding balance
S --> C : balance due
C --> UI : balance must be settled first
UI --> A : show balance due
end
deactivate S
deactivate C
deactivate RO
deactivate UI
deactivate A
@enduml
