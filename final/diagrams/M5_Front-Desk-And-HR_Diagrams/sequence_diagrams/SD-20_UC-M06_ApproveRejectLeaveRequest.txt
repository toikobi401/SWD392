@startuml
title SD-20 UC-M06 Approve / Reject Leave Request

actor "Manager" as A
boundary "ApprovalsPage" as UI
boundary "LeaveController" as C
entity "LeaveRequest" as L
database "MongoDB" as DB

A -> UI : open pending leave requests
activate A
activate UI
UI -> C : request pending leave requests
activate C
C -> L : get pending requests
activate L
L -> DB : find pending requests
activate DB
DB --> L : requests
deactivate DB
L --> C : pending requests
deactivate L
C --> UI : pending leave list
UI --> A : show pending list
A -> UI : approve request
UI -> C : submit decision
C -> C : validate request
alt employee has enough leave balance
C -> L : approve leave request
activate L
L -> DB : save decision
activate DB
DB --> L : decision saved
deactivate DB
L --> C : updated request
deactivate L
C --> UI : request approved
UI --> A : update pending list
else not enough leave balance
C --> UI : insufficient leave balance
UI --> A : show warning
end
deactivate C
deactivate UI
deactivate A
@enduml
