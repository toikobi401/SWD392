class StaffBookingPage <<boundary>> {
    +checkOut(bookingId)
}

class FrontDeskController <<boundary>> {
    +checkOut(req, res)
    +validateRequest(req)
}

class Folio <<entity>> {
    - _id : ObjectId
    - bookingId : ObjectId
    - status : FolioStatus
    - totalCharges : Number
    - totalPayments : Number
    - balance : Number
    - depositHeld : Number
    - closedAt : Date
    +getFolio(bookingId) : Folio
    +close(id) : Folio
}

class Room <<entity>> {
    - _id : ObjectId
    - status : RoomStatus
    - version : Number
    +release(roomId) : Room
}

enum FolioStatus <<enumeration>> {
    OPEN
    CLOSED
}

enum RoomStatus <<enumeration>> {
    VACANT_CLEAN
    VACANT_DIRTY
    OCCUPIED
    INSPECTED
    OUT_OF_ORDER
}
