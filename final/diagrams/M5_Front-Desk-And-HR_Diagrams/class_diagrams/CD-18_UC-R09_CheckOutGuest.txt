@startuml
title CD-18 UC-R09 Check Out Guest

skinparam classAttributeIconSize 0
hide empty members

class StaffBookingPage <<boundary>> {
    +checkOut(bookingId)
}

class Router <<boundary>> {
    +route(POST /api/front-desk/check-out)
    +authenticate(req)
    +requirePermission(permission)
    +validateBody(rules)
}

class FrontDeskController <<boundary>> {
    +checkOut(req, res)
}

class CheckOutService <<control>> {
    +checkOut(command) : CheckOutResult
}

class Folio <<entity>> {
    - _id : ObjectId
    - bookingId : ObjectId
    - status : FolioStatus
    - totalCharges : Number
    - totalPayments : Number
    - balance : Number
    - depositHeld : Number
    - closedAt : Date
    +getFolio(bookingId) : Folio
    +close(id) : Folio
}

class Room <<entity>> {
    - _id : ObjectId
    - status : RoomStatus
    - version : Number
    +release(roomId) : Room
}

enum FolioStatus <<enumeration>> {
    OPEN
    CLOSED
}

enum RoomStatus <<enumeration>> {
    VACANT_CLEAN
    VACANT_DIRTY
    OCCUPIED
    INSPECTED
    OUT_OF_ORDER
}

StaffBookingPage ..> Router : POST /api/front-desk/check-out
Router ..> FrontDeskController
FrontDeskController ..> CheckOutService
CheckOutService ..> Folio
CheckOutService ..> Room
Folio --> FolioStatus
Room --> RoomStatus

@enduml
