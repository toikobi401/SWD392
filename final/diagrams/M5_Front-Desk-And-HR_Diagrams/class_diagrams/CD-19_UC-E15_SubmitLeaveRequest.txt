skinparam classAttributeIconSize 0
hide empty members

class MyLeavePage <<boundary>> {
    +submitLeave(type, fromDate, toDate, reason)
}

class Router <<boundary>> {
    +route(POST /api/leave-requests)
    +authenticate(req)
}

class LeaveController <<boundary>> {
    +submit(req, res)
}

class LeaveApprovalService <<control>> {
    +submit(command) : LeaveRequest
}

class LeaveRequest <<entity>> {
    - _id : ObjectId
    - employeeId : ObjectId
    - type : LeaveType
    - fromDate : Date
    - toDate : Date
    - days : Number
    - reason : String
    - status : LeaveStatus
    - createdAt : Date
    +createRequest(data) : LeaveRequest
}

enum LeaveType <<enumeration>> {
    ANNUAL
    SICK
    UNPAID
    MATERNITY
}

enum LeaveStatus <<enumeration>> {
    PENDING
    APPROVED
    REJECTED
    CANCELLED
    ESCALATED
}

MyLeavePage ..> Router : POST /api/leave-requests
Router ..> LeaveController
LeaveController ..> LeaveApprovalService
LeaveApprovalService ..> LeaveRequest
LeaveRequest --> LeaveType
LeaveRequest --> LeaveStatus

@enduml
