    - employeeId : ObjectId
    - type : LeaveType
    - fromDate : Date
    - toDate : Date
    - days : Number

    ANNUAL
    SICK
    UNPAID
    MATERNITY
}

    PENDING
    APPROVED
    REJECTED
    CANCELLED
    ESCALATED
}
