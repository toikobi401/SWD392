class ApprovalsPage <<boundary>> {
    +loadPendingLeave()
    +decide(id, approve, note)

class LeaveController <<boundary>> {
    +pending(req, res)
    +decide(req, res)
    +validateRequest(req)
}

class LeaveRequest <<entity>> {
    - _id : ObjectId
    - employeeId : ObjectId
    - type : LeaveType
    - fromDate : Date
    - toDate : Date
    - days : Number
    - reason : String
    - status : LeaveStatus
    - approverId : ObjectId
    - decidedAt : Date
    - decisionNote : String
    - coverageOverrideJustification : String
    - createdAt : Date
    +getPending(department) : LeaveRequest[]
    +approve(id) : LeaveRequest
}

enum LeaveType <<enumeration>> {
    ANNUAL
    SICK
    UNPAID
    MATERNITY
}

enum LeaveStatus <<enumeration>> {
    PENDING
    APPROVED
    REJECTED
    CANCELLED
    ESCALATED
}
