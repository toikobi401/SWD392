@startuml
title CD-20 UC-M06 Approve / Reject Leave Request

skinparam classAttributeIconSize 0
hide empty members

class ApprovalsPage <<boundary>> {
    +loadPendingLeave()
    +decide(id, approve, note)
}

class Router <<boundary>> {
    +route(GET /api/leave-requests/pending)
    +route(PATCH /api/leave-requests/:id/decision)
    +authenticate(req)
    +requirePermission(permission)
}

class LeaveController <<boundary>> {
    +pending(req, res)
    +decide(req, res)
}

class LeaveApprovalService <<control>> {
    +pendingQueue(managerId, department) : LeaveRequest[]
    +decide(command) : LeaveRequest
}

class LeaveRequest <<entity>> {
    - _id : ObjectId
    - employeeId : ObjectId
    - type : LeaveType
    - fromDate : Date
    - toDate : Date
    - days : Number
    - reason : String
    - status : LeaveStatus
    - approverId : ObjectId
    - decidedAt : Date
    - decisionNote : String
    - coverageOverrideJustification : String
    - createdAt : Date
    +getPending(department) : LeaveRequest[]
    +approve(id) : LeaveRequest
}

enum LeaveType <<enumeration>> {
    ANNUAL
    SICK
    UNPAID
    MATERNITY
}

enum LeaveStatus <<enumeration>> {
    PENDING
    APPROVED
    REJECTED
    CANCELLED
    ESCALATED
}

ApprovalsPage ..> Router : GET /api/leave-requests/pending\nPATCH /api/leave-requests/:id/decision
Router ..> LeaveController
LeaveController ..> LeaveApprovalService
LeaveApprovalService ..> LeaveRequest
LeaveRequest --> LeaveType
LeaveRequest --> LeaveStatus

@enduml
