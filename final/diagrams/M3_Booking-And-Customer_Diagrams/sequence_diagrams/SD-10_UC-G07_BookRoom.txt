@startuml
title SD-10 UC-G07 Book Room

actor "Guest" as G
boundary "BookingWizardPage" as UI
boundary "BookingController" as C
entity "Booking" as B
database "MongoDB" as DB
participant "payOS" as X

G -> UI : confirm booking details
activate G
activate UI
UI -> C : submit booking
activate C
C -> C : validate request
C -> B : check room availability
activate B
B -> DB : count overlapping bookings
activate DB
DB --> B : booking counts
deactivate DB
B --> C : availability
deactivate B
alt rooms available
C -> B : create pending booking
activate B
B -> DB : save booking
activate DB
DB --> B : booking saved
deactivate DB
B --> C : booking
deactivate B
C -> X : create payment link
activate X
X --> C : checkout URL
deactivate X
C --> UI : booking pending and payment URL
UI --> G : redirect to payOS checkout
else rooms no longer available
C --> UI : no availability
UI --> G : show error and search again
end
deactivate C
deactivate UI
deactivate G
@enduml
