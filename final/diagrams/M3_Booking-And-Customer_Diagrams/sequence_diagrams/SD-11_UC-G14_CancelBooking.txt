@startuml
title SD-11 UC-G14 Cancel Booking

actor "Customer" as CU
boundary "BookingDetailPage" as UI
boundary "BookingController" as C
entity "Booking" as B
database "MongoDB" as DB
participant "Email service" as E

note over CU, UI
    Precondition: booking detail is already loaded (UC-C12).
    BookingDetailPage and its CancelDialog are in CustomerPages.tsx.
end note

CU -> UI : setConfirming(true)
activate CU
activate UI
alt booking is not CONFIRMED
    UI --> CU : cancellation action is unavailable
else confirmed booking
    UI --> CU : show policy and confirmation dialog with optional reason
    opt Customer keeps booking
        CU -> UI : setConfirming(false)
        UI --> CU : close dialog; no request sent
    end
end
deactivate UI
deactivate CU

CU -> UI : cancel.mutate(reason)
activate CU
activate UI
UI -> C : POST /api/bookings/:id/cancel (reason)\ncancel(req, res)
activate C
C -> C : cancel(req, res)\nauthenticate; requirePermission(BOOKING_CANCEL)

alt missing, expired or invalid access token
    C --> UI : 401 UNAUTHORIZED / TOKEN_EXPIRED / INVALID_TOKEN
    UI --> CU : show authentication error
else missing BOOKING_CANCEL permission
    C --> UI : 403 FORBIDDEN
    UI --> CU : show access error
else authorized request
    C -> B : findById(bookingId)
    activate B
    B -> DB : query booking by _id
    activate DB
    DB --> B : booking document or null
    deactivate DB
    B --> C : booking or null
    deactivate B
    C -> C : cancel(req, res)\ncheck ownership; BOOKING_MODIFY permits staff override
    alt booking missing or belongs to another customer
        C --> UI : 404 BOOKING_NOT_FOUND
        UI --> CU : show booking not found
    else booking found and accessible
        C -> C : cancel(req, res)\ncan(status, CANCEL); evaluate(booking, now, booking.nonRefundable)
        alt status is not CONFIRMED (checked in, already cancelled or other state)
            C --> UI : 409 CANNOT_CANCEL
            UI --> CU : show lifecycle error; contact front desk for an active stay
        else cancellation allowed
            alt nonRefundable rate
                C -> C : cancel(req, res)\npenalty = totalAmount; refundable = 0; requiresApproval = false
            else at least 48 hours before check-in
                C -> C : cancel(req, res)\npenalty = 0; refundable = totalAmount; requiresApproval = false
            else at least 24 but less than 48 hours
                C -> C : cancel(req, res)\npenalty = round(totalAmount / 2);\nrefundable = totalAmount - penalty; requiresApproval = false
            else less than 24 hours before check-in
                C -> C : cancel(req, res)\npenalty = totalAmount; refundable = 0; requiresApproval = true
            end
            C -> C : cancel(req, res)\nnext(status, CANCEL); set cancelledAt and cancellationReason
            C -> B : save()
            activate B
            B -> DB : persist status CANCELLED, cancelledAt and cancellationReason
            activate DB
            DB --> B : saved document
            deactivate DB
            B --> C : cancelled booking
            deactivate B
            C -> C : cancel(req, res)\nCANCELLED no longer consumes availability; refundStatus = NONE
            alt refundable > 0
                C -> C : cancel(req, res)\nrefund(bookingId, refundable, actorId): check captured total;\npayOS refund proxy returns NOT_SUPPORTED locally;\nrecord failed reversal and flag manual bank transfer
            else requiresApproval
                C -> C : cancel(req, res)\ncreateRefundRequestForApproval: create PENDING request if PAID payment exists;\nrefundStatus = AWAITING_APPROVAL
            else non-refundable rate
                C -> C : cancel(req, res)\nkeep refundStatus = NONE; no refund
            end
            alt refundable exceeds captured total
                C --> UI : 400 REFUND_EXCEEDS_CAPTURE (booking remains CANCELLED)
                UI --> CU : show refund error
            else refund handling completed
                C -> E : POST /send\nsend(message): cancellation notice to guest.email
                activate E
                alt email accepted
                    E --> C : delivery accepted
                else delivery failure
                    E --> C : delivery error
                    C -> C : cancel(req, res)\ncatch and log notification failure; keep cancellation
                end
                deactivate E
                C -> C : cancel(req, res)\nrecord BOOKING_CANCELLED with penalty and refundable
                C --> UI : 200 OK (bookingCode, status: CANCELLED, refundable,\nrequiresApproval, refundStatus)
                UI --> CU : show cancellation/refund outcome; close dialog;\ninvalidate booking and my-bookings queries
            end
        end
    end
end

deactivate C
deactivate UI
deactivate CU

note over C, E
    Only this room is cancelled; sibling bookings remain unchanged.
    Service, policy, payment, refund-request and audit work is folded into C.
    No payOS refund network call exists; bank transfer is manual.
    Email assumes NOTIFICATION_API_URL is configured; development can log mail.
    The code logs delivery failure; no durable retry queue is implemented.
    The saved status is CANCELLED, including cases awaiting refund approval.
end note
@enduml
