@startuml
title SD-12 UC-G13 Check Booking Status

actor "Guest" as G
boundary "BookingLookupPage" as UI
boundary "BookingController" as C
entity "Booking" as B
database "MongoDB" as DB

G -> UI : enter booking code and email
activate G
activate UI
UI -> C : look up booking
activate C
C -> C : validate request
C -> B : find booking by code and email
activate B
B -> DB : query booking
activate DB
DB --> B : search result
deactivate DB
B --> C : booking
deactivate B
alt booking found
C --> UI : booking details and status
UI --> G : show booking status
else no matching booking
C --> UI : booking not found
UI --> G : show error message
end
deactivate C
deactivate UI
deactivate G
@enduml
