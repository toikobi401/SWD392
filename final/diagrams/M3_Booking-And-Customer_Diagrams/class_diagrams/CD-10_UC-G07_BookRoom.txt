@startuml
title CD-10 UC-G07 Book Room

skinparam classAttributeIconSize 0
hide empty members

class BookingWizardPage <<boundary>> {
    +submitBooking()
}

class Router <<boundary>> {
    +route(POST /api/bookings)
    +optionalAuth(req)
    +validateBody(rules)
}

class BookingController <<boundary>> {
    +create(req, res)
}

class BookingService <<control>> {
    +bookRoom(command) : BookingResult
}

class "Booking" as B <<entity>> {
    - _id : ObjectId
    - bookingCode : String
    - reservationCode : String
    - customerId : ObjectId
    - guest.fullName : String
    - guest.email : String
    - guest.phone : String
    - guest.specialRequest : String
    - occupantName : String
    - roomTypeId : ObjectId
    - checkInDate : Date
    - checkOutDate : Date
    - adults : Number
    - children : Number
    - status : BookingStatus
    - addOnServices : {serviceId: ObjectId, quantity: Number, price: Number}[]
    - promotionId : ObjectId
    - subtotal : Number
    - discount : Number
    - tax : Number
    - totalAmount : Number
    - cancelledAt : Date
    - cancellationReason : String
    +checkAvailability(typeId, dates) : Boolean
    +createPending(data) : Booking
}

enum BookingStatus <<enumeration>> {
    PENDING
    CONFIRMED
    CHECKED_IN
    CHECKED_OUT
    CANCELLED
    NO_SHOW
}

class "PayOS" as P <<external>> {
    +createPaymentLink(order) : String
}
