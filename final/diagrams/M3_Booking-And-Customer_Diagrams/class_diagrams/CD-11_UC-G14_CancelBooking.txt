@startuml
title CD-11 UC-G14 Cancel Booking

skinparam classAttributeIconSize 0
hide empty members

class BookingDetailPage <<boundary>> {
    +cancelBooking(bookingId)
}

class Router <<boundary>> {
    +route(POST /api/bookings/:id/cancel)
    +authenticate(req)
    +requirePermission(permission)
}

class BookingController <<boundary>> {
    +cancel(req, res)
}

class BookingService <<control>> {
    +cancelBooking(command) : CancellationResult
}

class "Booking" as B <<entity>> {
    - _id : ObjectId
    - bookingCode : String
    - customerId : ObjectId
    - guest.fullName : String
    - guest.email : String
    - checkInDate : Date
    - status : BookingStatus
    - totalAmount : Number
    - nonRefundable : Boolean
    - cancelledAt : Date
    - cancellationReason : String
    +getBooking(id) : Booking
    +cancel(id) : Booking
}

enum BookingStatus <<enumeration>> {
    PENDING
    CONFIRMED
    CHECKED_IN
    CHECKED_OUT
    CANCELLED
    NO_SHOW
}

BookingDetailPage ..> Router : POST /api/bookings/:id/cancel
Router ..> BookingController
BookingController ..> BookingService
BookingService ..> B
B --> BookingStatus : status

@enduml
