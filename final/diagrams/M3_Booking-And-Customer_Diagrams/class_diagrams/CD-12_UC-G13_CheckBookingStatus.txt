class BookingLookupPage <<boundary>> {
    +lookup(code, email)
}

class Router <<boundary>> {
    +route(GET /api/bookings/lookup)
}

class BookingController <<boundary>> {
    +lookup(req, res)
}

class BookingQueries <<control>> {
    +lookup(code, email) : Reservation
}

class "Booking" as B <<entity>> {
    - _id : ObjectId
    - bookingCode : String
    - reservationCode : String
    - guest.fullName : String
    - guest.email : String
    - occupantName : String
    - roomTypeId : ObjectId
    - checkInDate : Date
    - checkOutDate : Date
    - adults : Number
    - children : Number
    - status : BookingStatus
    - totalAmount : Number
    +findByCodeAndEmail(code, email) : Booking
}

enum BookingStatus <<enumeration>> {
    PENDING
    CONFIRMED
    CHECKED_IN
    CHECKED_OUT
    CANCELLED
    NO_SHOW
}

BookingLookupPage ..> Router : GET /api/bookings/lookup
Router ..> BookingController
BookingController ..> BookingQueries
BookingQueries ..> B
B --> BookingStatus : status
