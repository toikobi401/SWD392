class "BookingLookupPage" as UI <<boundary>> {
    + setQuery(query)
}

class "BookingController" as C <<boundary>> {
    + lookup(req, res)
}

class "Booking" as B <<entity>> {
    - _id : ObjectId
    - bookingCode : String
    - reservationCode : String
    - guest.fullName : String
    - guest.email : String
    - occupantName : String
    - roomTypeId : ObjectId
    - checkInDate : Date
    - checkOutDate : Date
    - adults : Number
    - children : Number
    - status : BookingStatus
    - totalAmount : Number
    + findOne(filter) : Booking
    + find(filter) : Query<Booking[]>
    + populate(path) : Query<Booking[]>
    + sort(order) : Query<Booking[]>
}

enum BookingStatus <<enumeration>> {
    PENDING
    CONFIRMED
    CHECKED_IN
    CHECKED_OUT
    CANCELLED
    NO_SHOW
}

note bottom of B
    populate and sort are Mongoose query-chain operations.
    RoomType population is summarized within the Booking query.
end note

note right of C
    Controller self-messages summarize existing
    service/rule/middleware orchestration.
