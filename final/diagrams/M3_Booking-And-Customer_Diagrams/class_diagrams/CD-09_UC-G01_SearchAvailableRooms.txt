class RoomSearchPage <<boundary>> {
    +search(criteria)
}

class Router <<boundary>> {
    +route(GET /api/rooms/search)
}

class RoomController <<boundary>> {
    +search(req, res)
}

class RoomQueries <<control>> {
    +searchPriced(criteria) : PricedAvailability[]
}

class "RoomType" as RT <<entity>> {
    - _id : ObjectId
    - name : String
    - description : String
    - capacity : Number
    - basePrice : Number
    - amenities : String[]
    - images : String[]
    - totalRooms : Number
    - seasonalRates : {from: Date, to: Date, price: Number}[]
    - isActive : Boolean
    +getActiveTypes() : RoomType[]
}

class "Booking" as B <<entity>> {
    - _id : ObjectId
    - roomTypeId : ObjectId
    - checkInDate : Date
    - checkOutDate : Date
    - status : BookingStatus
    +countOverlapping(typeId, dates) : Number
}

enum BookingStatus <<enumeration>> {
    PENDING
    CONFIRMED
    CHECKED_IN
    CHECKED_OUT
    CANCELLED
    NO_SHOW
}

RoomSearchPage ..> Router : GET /api/rooms/search
Router ..> RoomController
RoomController ..> RoomQueries
RoomQueries ..> RT
RoomQueries ..> B
RT "1" -- "0..*" B : roomTypeId
B --> BookingStatus : status
